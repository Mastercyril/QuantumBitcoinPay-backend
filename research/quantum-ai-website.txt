       const GEMINI_API_URL = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.0-flash:generateContent?key=${GEMINI_API_KEY}`;

       // Function to show the modal
       function showModal(title = "AI Response ✨") {
           if (!aiResponseModal) return;
           modalTitle.textContent = title;
           modalBody.innerHTML = ""; // Clear previous content
           modalLoading.classList.remove('hidden');
           aiResponseModal.classList.remove('hidden');
       }

       // Function to hide the modal
       function hideModal() {
           if (aiResponseModal) {
               aiResponseModal.classList.add('hidden');
               modalLoading.classList.add('hidden');
           }
       }
       if (closeModalBtn) {
           closeModalBtn.addEventListener('click', hideModal);
       }
       // Hide modal if clicked outside of modal-content

           aiResponseModal.addEventListener('click', (event) => {

       // Function to call Gemini API
       async function callGeminiAPI(promptText) {
           showModal("QuantumAI is thinking...");
           modalBody.innerHTML = ""; // Clear previous content
           modalLoading.classList.remove('hidden');

           const payload = {
               contents: [{ role: "user", parts: [{ text: promptText }] }]
           };

           try {
               const response = await fetch(GEMINI_API_URL, {
                   method: 'POST',
                   headers: { 'Content-Type': 'application/json' },
                   body: JSON.stringify(payload)
               });

               if (!response.ok) {
                   const errorData = await response.json();
                   console.error("API Error:", errorData);
                   throw new Error(`API request failed with status ${response.status}: ${errorData?.error?.message || 'Unknown error'}`);
               }

               const result = await response.json();
               modalLoading.classList.add('hidden');

               if (result.candidates && result.candidates.length > 0 &&
                   result.candidates[0].content && result.candidates[0].content.parts &&
                   result.candidates[0].content.parts.length > 0) {
                   const text = result.candidates[0].content.parts[0].text;
                   modalBody.innerHTML = text.replace(/\n/g, '<br>'); // Basic formatting
               } else {
                   console.error("Unexpected API response structure:", result);
                   modalBody.textContent = "Sorry, I couldn't get a clear response. The AI might be busy or the response was empty.";
               }
           } catch (error) {
               console.error("Error calling Gemini API:", error);
               modalLoading.classList.add('hidden');
               modalBody.textContent = `An error occurred: ${error.message}. Please try again later. Check console for details.`;
           }
