"""
Quantum AI Bridge Server
Allows remote control of local AI system via secure API
"""
from fastapi import FastAPI, HTTPException, Header, WebSocket
from fastapi.middleware.cors import CORSMiddleware
from pydantic import BaseModel
import subprocess
import json
import os
import sys
from pathlib import Path
from typing import Optional
import hashlib
import secrets
import uvicorn
from datetime import datetime

# Configuration
CONFIG_FILE = "bridge_config.json"
MEMORY_FILE = "Q:/quantum_ai/memory/conversation_memory.txt"
Q_PERSONALITY_FILE = "Q:/quantum_ai/core/Q_personality.json"

app = FastAPI(title="Quantum AI Bridge", version="2.0")

# Enable CORS for web interfaces
app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

# Load or create configuration
def load_config():
    if os.path.exists(CONFIG_FILE):
        with open(CONFIG_FILE, 'r') as f:
            return json.load(f)
    else:
        api_key = secrets.token_urlsafe(32)
        config = {
            "api_key": hashlib.sha256(api_key.encode()).hexdigest(),
            "q_drive_path": "Q:/quantum_ai",
            "allowed_commands": ["python", "pip", "ollama"],
            "port": 8765
        }
        with open(CONFIG_FILE, 'w') as f:
            json.dump(config, f, indent=2)
        print(f"\n🔑 Your API Key (save this!): {api_key}\n")
        return config

config = load_config()

# Security: API Key verification
def verify_api_key(x_api_key: str = Header(...)):
    if hashlib.sha256(x_api_key.encode()).hexdigest() != config['api_key']:
        raise HTTPException(status_code=401, detail="Invalid API key")
    return True

# Models
class CommandRequest(BaseModel):
    command: str
    args: list = []
    working_directory: Optional[str] = None

class FileOperation(BaseModel):
    operation: str  # read, write, delete, list
    path: str
    content: Optional[str] = None

class MemoryOperation(BaseModel):
    operation: str  # append, read, search
    content: Optional[str] = None
    query: Optional[str] = None

# Endpoints
@app.get("/")
async def root():
    return {
        "system": "Quantum AI Bridge",
        "status": "active",
        "version": "2.0",
        "q_personality": "loaded" if os.path.exists(Q_PERSONALITY_FILE) else "not_found",
        "timestamp": datetime.now().isoformat()
    }

@app.get("/health")
async def health_check():
    q_drive_exists = os.path.exists(config['q_drive_path'])
    return {
        "status": "healthy",
        "q_drive_mounted": q_drive_exists,
        "memory_accessible": os.path.exists(MEMORY_FILE) if q_drive_exists else False
    }

@app.post("/execute")
async def execute_command(cmd: CommandRequest, authorized: bool = None):
    """Execute system commands (Python scripts, pip installs, etc.)"""
    if cmd.command not in config['allowed_commands']:
        raise HTTPException(status_code=403, detail="Command not allowed")
    
    try:
        working_dir = cmd.working_directory or config['q_drive_path']
        result = subprocess.run(
            [cmd.command] + cmd.args,
            cwd=working_dir,
            capture_output=True,
            text=True,
            timeout=300
        )
        
        return {
            "success": result.returncode == 0,
            "stdout": result.stdout,
            "stderr": result.stderr,
            "return_code": result.returncode
        }
    except subprocess.TimeoutExpired:
        raise HTTPException(status_code=408, detail="Command timeout")
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

@app.post("/files")
async def file_operations(file_op: FileOperation, authorized: bool = None):
    """Handle file operations on Q drive"""
    full_path = os.path.join(config['q_drive_path'], file_op.path)
    
    try:
        if file_op.operation == "read":
            with open(full_path, 'r', encoding='utf-8') as f:
                return {"content": f.read()}
        
        elif file_op.operation == "write":
            os.makedirs(os.path.dirname(full_path), exist_ok=True)
            with open(full_path, 'w', encoding='utf-8') as f:
                f.write(file_op.content)
            return {"success": True, "path": full_path}
        
        elif file_op.operation == "delete":
            os.remove(full_path)
            return {"success": True}
        
        elif file_op.operation == "list":
            items = os.listdir(full_path)
            return {"items": items}
        
        else:
            raise HTTPException(status_code=400, detail="Invalid operation")
            
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

@app.post("/memory")
async def memory_operations(mem_op: MemoryOperation, authorized: bool = None):
    """Manage conversation memory"""
    try:
        if mem_op.operation == "append":
            timestamp = datetime.now().isoformat()
            entry = f"\n[{timestamp}]\n{mem_op.content}\n{'='*80}\n"
            with open(MEMORY_FILE, 'a', encoding='utf-8') as f:
                f.write(entry)
            return {"success": True}
        
        elif mem_op.operation == "read":
            with open(MEMORY_FILE, 'r', encoding='utf-8') as f:
                return {"content": f.read()}
        
        elif mem_op.operation == "search":
            with open(MEMORY_FILE, 'r', encoding='utf-8') as f:
                content = f.read()
            matches = [line for line in content.split('\n') if mem_op.query.lower() in line.lower()]
            return {"matches": matches[:50]}  # Return top 50 matches
            
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

@app.get("/q-personality")
async def get_q_personality(authorized: bool = None):
    """Retrieve Q personality configuration"""
    try:
        with open(Q_PERSONALITY_FILE, 'r', encoding='utf-8') as f:
            return json.load(f)
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

@app.websocket("/ws")
async def websocket_endpoint(websocket: WebSocket):
    """WebSocket for real-time communication"""
    await websocket.accept()
    try:
        while True:
            data = await websocket.receive_text()
            message = json.loads(data)
            
            # Process message and send response
            response = {
                "type": "acknowledgment",
                "message": f"Received: {message.get('command', 'unknown')}",
                "timestamp": datetime.now().isoformat()
            }
            await websocket.send_text(json.dumps(response))
    except Exception as e:
        print(f"WebSocket error: {e}")

if __name__ == "__main__":
    print("🌉 Starting Quantum AI Bridge Server...")
    print(f"🔗 API will be available at: http://localhost:{config['port']}")
    print(f"📂 Q Drive Path: {config['q_drive_path']}")
    uvicorn.run(app, host="127.0.0.1", port=config['port'])

Auto-Setup Script (auto_setup.py)

"""
Quantum AI Auto-Setup Script
Automatically configures the system when Q drive is connected
"""
import os
import sys
import json
import subprocess
from pathlib import Path

class QuantumSetup:
    def __init__(self):
        self.drive_letter = self.detect_q_drive()
        self.base_path = Path(f"{self.drive_letter}:/quantum_ai")
        
    def detect_q_drive(self):
        """Detect the 2TB drive (looks for quantum_ai folder)"""
        for letter in 'DEFGHIJKLMNOPQRSTUVWXYZ':
            path = Path(f"{letter}:/quantum_ai")
            if path.exists():
                return letter
        return None
    
    def verify_structure(self):
        """Ensure all required directories exist"""
        required_dirs = [
            "core", "memory", "models", "scripts", 
            "models/local_llm", "models/embeddings"
        ]
        for dir_name in required_dirs:
            dir_path = self.base_path / dir_name
            dir_path.mkdir(parents=True, exist_ok=True)
        print("✅ Directory structure verified")
    
    def initialize_memory(self):
        """Create memory file if it doesn't exist"""
        memory_file = self.base_path / "memory/conversation_memory.txt"
        if not memory_file.exists():
            with open(memory_file, 'w', encoding='utf-8') as f:
                f.write("=== Quantum AI Memory System Initialized ===\n")
                f.write(f"Timestamp: {self.get_timestamp()}\n")
                f.write("="*80 + "\n\n")
        print("✅ Memory system initialized")
    
    def check_python_packages(self):
        """Verify required packages are installed"""
        required = ['fastapi', 'uvicorn', 'pydantic', 'websockets']
        missing = []
        
        for package in required:
            try:
                __import__(package)
            except ImportError:
                missing.append(package)
        
        if missing:
            print(f"⚠️  Missing packages: {', '.join(missing)}")
            print("Run: pip install -r requirements.txt")
        else:
            print("✅ All Python packages installed")
    
    def create_desktop_shortcut(self):
        """Create shortcut to launch bridge server"""
        desktop = Path.home() / "Desktop"
        shortcut_path = desktop / "Start_Quantum_Bridge.bat"
        
        content = f"""@echo off
cd /d {self.base_path}
python bridge_server.py
pause
"""
        with open(shortcut_path, 'w') as f:
            f.write(content)
        print("✅ Desktop shortcut created")
    
    @staticmethod
    def get_timestamp():
        from datetime import datetime
        return datetime.now().isoformat()
    
    def run(self):
        print("\n⚛️  Quantum AI Setup Starting...\n")
        
        if not self.drive_letter:
            print("❌ Q drive not found! Please connect your 2TB drive.")
            sys.exit(1)
        
        print(f"🔍 Found quantum_ai at {self.drive_letter}:/quantum_ai")
        
        self.verify_structure()
        self.initialize_memory()
        self.check_python_packages()
        self.create_desktop_shortcut()
        
        print("\n✨ Setup Complete!")
        print(f"\n📍 Your Q drive: {self.drive_letter}:")
        print("🚀 Launch bridge: python bridge_server.py")
        print("or double-click: Start_Quantum_Bridge.bat on desktop\n")

if __name__ == "__main__":
    setup = QuantumSetup()
    setup.run()

🔗 Component 3: Remote Connection Client

This is how Perplexity (or any remote interface) connects to your system.

"""
Quantum AI Remote Client
For Perplexity or other remote interfaces to connect
"""
import requests
import json
from typing import Dict, Any

class QuantumBridgeClient:
    def __init__(self, base_url: str, api_key: str):
        self.base_url = base_url
        self.headers = {"X-API-Key": api_key}
    
    def health_check(self) -> Dict[str, Any]:
        """Check if bridge is accessible"""
        response = requests.get(f"{self.base_url}/health", headers=self.headers)
        return response.json()
    
    def execute_command(self, command: str, args: list = []) -> Dict[str, Any]:
        """Execute a command on remote system"""
        payload = {"command": command, "args": args}
        response = requests.post(
            f"{self.base_url}/execute", 
            json=payload, 
            headers=self.headers
        )
        return response.json()
    
    def read_file(self, path: str) -> str:
        """Read file from Q drive"""
        payload = {"operation": "read", "path": path}
        response = requests.post(
            f"{self.base_url}/files", 
            json=payload, 
            headers=self.headers
        )
        return response.json()['content']
    
    def write_file(self, path: str, content: str) -> Dict[str, Any]:
        """Write file to Q drive"""
        payload = {"operation": "write", "path": path, "content": content}
        response = requests.post(
            f"{self.base_url}/files", 
            json=payload, 
            headers=self.headers
        )
        return response.json()
    
    def append_memory(self, content: str) -> Dict[str, Any]:
        """Add to conversation memory"""
        payload = {"operation": "append", "content": content}
        response = requests.post(
            f"{self.base_url}/memory", 
            json=payload, 
            headers=self.headers
        )
        return response.json()
    
    def get_q_personality(self) -> Dict[str, Any]:
        """Retrieve Q personality config"""
        response = requests.get(
            f"{self.base_url}/q-personality", 
            headers=self.headers
        )
        return response.json()

python bridge_server.py
